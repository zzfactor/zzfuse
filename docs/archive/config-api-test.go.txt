const (
	// Default API base URL - can be overridden with TRAZR_BASE_URL env var
	defaultBaseURL = "http://localhost:18888"
	
	// Test credentials - these should be set as environment variables
	testUsername = "testuser"
	testPassword = "testpass"
)

// TestClient handles authentication and API requests
type TestClient struct {
	baseURL string
	token   string
	client  *http.Client
}

// NewTestClient creates a new test client with authentication
func NewTestClient() (*TestClient, error) {
	baseURL := os.Getenv("TRAZR_BASE_URL")
	if baseURL == "" {
		baseURL = defaultBaseURL
	}

	client := &TestClient{
		baseURL: baseURL,
		client: &http.Client{
			Timeout: 30 * time.Second,
		},
	}

	// Get auth token using test credentials
	if err := client.authenticate(); err != nil {
		return nil, fmt.Errorf("authentication failed: %w", err)
	}

	return client, nil
}

// authenticate gets an auth token using test credentials
func (c *TestClient) authenticate() error {
	// Check if token is provided via environment variable first
	if token := os.Getenv("TRAZR_AUTH_TOKEN"); token != "" {
		c.token = token
		return nil
	}

	// Otherwise, authenticate using username/password
	authPayload := map[string]string{
		"username": testUsername,
		"password": testPassword,
	}

	payloadBytes, err := json.Marshal(authPayload)
	if err != nil {
		return fmt.Errorf("failed to marshal auth payload: %w", err)
	}

	resp, err := c.client.Post(c.baseURL+"/auth/login", "application/json", bytes.NewBuffer(payloadBytes))
	if err != nil {
		return fmt.Errorf("auth request failed: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return fmt.Errorf("auth failed with status %d", resp.StatusCode)
	}

	var authResp struct {
		Token string `json:"token"`
	}

	if err := json.NewDecoder(resp.Body).Decode(&authResp); err != nil {
		return fmt.Errorf("failed to decode auth response: %w", err)
	}

	c.token = authResp.Token
	return nil
}

// makeRequest makes an authenticated HTTP request
func (c *TestClient) makeRequest(method, path string, body interface{}) (*http.Response, error) {
	var reqBody io.Reader
	if body != nil {
		bodyBytes, err := json.Marshal(body)
		if err != nil {
			return nil, fmt.Errorf("failed to marshal request body: %w", err)
		}
		reqBody = bytes.NewBuffer(bodyBytes)
	}

	req, err := http.NewRequest(method, c.baseURL+path, reqBody)
	if err != nil {
		return nil, fmt.Errorf("failed to create request: %w", err)
	}

	req.Header.Set("Authorization", "Bearer "+c.token)
	if body != nil {
		req.Header.Set("Content-Type", "application/json")
	}

	return c.client.Do(req)
}

	}
	defer resp.Body.Close()

	// Accept both 201 (created) and 409 (conflict if already exists)
	if resp.StatusCode != http.StatusCreated && resp.StatusCode != http.StatusConflict {
		body, _ := io.ReadAll(resp.Body)
		t.Errorf("Expected status 201 or 409, got %d. Response: %s", resp.StatusCode, string(body))
		return
	}

	if resp.StatusCode == http.StatusCreated {
		var response map[string]interface{}
		if err := json.NewDecoder(resp.Body).Decode(&response); err != nil {
			t.Fatalf("Failed to decode create response: %v", err)
		}

		// Verify response structure
		expectedFields := []string{"id", "name", "enabled", "created_at", "updated_at"}
		for _, field := range expectedFields {
			if _, exists := response[field]; !exists {
				t.Errorf("Missing expected field in create response: %s", field)
			}
		}

		t.Log("✓ Schema configuration created successfully")
	} else {
		t.Log("✓ Schema configuration already exists (expected for repeated tests)")
	}
}

// TestConfigAPIErrorHandling tests various error scenarios
func TestConfigAPIErrorHandling(t *testing.T) {
	client, err := NewTestClient()
	if err != nil {
		t.Skipf("Skipping integration test - failed to create client: %v", err)
	}

	testCases := []struct {
		name           string
		method         string
		path           string
		payload        interface{}
		expectedStatus int
	}{
		{"InvalidConfigType", "GET", "/api/v1/config/invalid", nil, http.StatusBadRequest},
		{"InvalidJSON", "POST", "/api/v1/config/schemas", "invalid-json", http.StatusBadRequest},
		{"NonExistentConfig", "GET", "/api/v1/config/schemas/99999", nil, http.StatusNotFound},
	}

	for _, tc := range testCases {
		t.Run(tc.name, func(t *testing.T) {
			var resp *http.Response
			var err error

			if tc.payload == "invalid-json" {
				// Send invalid JSON directly
				req, _ := http.NewRequest(tc.method, client.baseURL+tc.path, bytes.NewBufferString("invalid-json"))
				req.Header.Set("Authorization", "Bearer "+client.token)
				req.Header.Set("Content-Type", "application/json")
				resp, err = client.client.Do(req)
			} else {
				resp, err = client.makeRequest(tc.method, tc.path, tc.payload)
			}

			if err != nil {
				t.Fatalf("Error handling test request failed: %v", err)
			}
			defer resp.Body.Close()

			if resp.StatusCode != tc.expectedStatus {
				body, _ := io.ReadAll(resp.Body)
				t.Errorf("Expected status %d, got %d. Response: %s", tc.expectedStatus, resp.StatusCode, string(body))
			}
		})
	}

	t.Log("✓ Error handling working correctly")
}
